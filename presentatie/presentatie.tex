\begin{frame}{Probleemstelling: een didactische kloof}
  \begin{block}{Het knelpunt ligt niet enkel bij de taal, maar bij de didactiek}
    Moderne talen worden aangeleerd via interactieve sandboxes met directe feedback. COBOL-onderwijs blijft grotendeels statisch.
  \end{block}
  \begin{itemize}
    \item \textbf{Statische leermiddelen:} lijvige PDF-handleidingen en abstracte theorie.
    \item \textbf{Weinig visuele feedback:} het leerproces voelt droog en abstract.
    \item \textbf{Cognitieve dissonantie:} de strikte, verbose syntax botst met de flexibiliteit die junioren van Python gewend zijn.
  \end{itemize}
  \pause
  \centering\medskip
  \emph{Gevolg: talent haakt af nog voor het de logica van het mainframe beheerst.}
\end{frame}

  \end{enumerate}

\section{Gamification}

\begin{frame}{Gamification: drie kaders uit de literatuur}
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \begin{block}{PBL-triade}
        \footnotesize Points, Badges en Leaderboards. Het startpunt, niet het eindpunt \autocite{KevinWerbach2020}.
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{MDA}
        \footnotesize Mechanics, Dynamics en Aesthetics. Code en regels vertalen zich in beleving \autocite{Hunicke2004}.
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{Octalysis}
        \footnotesize Acht Core Drives. Human-Focused Design en de balans tussen White Hat en Black Hat \autocite{Chou2015}.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \footnotesize Gevaar bij alle drie: de \alert{PBL-fallacy}, de misvatting dat een dunne laag punten en badges een systeem vanzelf motiverend maakt \autocite{KevinWerbach2020,Chou2015}.
\end{frame}

\section{Proof-of-Concept}

\begin{frame}{Proof-of-Concept: Learn COBOL}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Een interactieve webapplicatie die COBOL aanleert via een gamified leerpad van zeven levels, gericht op junioren met een Python-achtergrond.
      \medskip
      \begin{block}{Kernkeuze: heuristische validatie}
        \footnotesize Geen volledige compiler, maar een checklist van leerdoelen per level. De feedback is direct en in dezelfde taal als de opdracht.
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Technologie}
      \begin{itemize}\small
        \item React, TypeScript en Vite
        \item Supabase (PostgreSQL en auth)
        \item Monaco-editor met COBOL-highlighting
        \item gehost op Vercel
      \end{itemize}
      \medskip
      \footnotesize Naast de COBOL-editor staat steeds een Python-referentie, zodat onbekende syntax aan een vertrouwd concept gekoppeld wordt.
    \end{column}
  \end{columns}
\end{frame}

%==============================================================================
\begin{frame}{Het leerpad: zeven levels}
  \centering
  \small
  \begin{tabularx}{0.92\textwidth}{@{}c X@{}}
    \toprule
    \textbf{Level} & \textbf{Onderwerp} \\
    \midrule
    1 & Vier divisions en minimale programmastructuur \\
    2 & \texttt{PIC}-clausules: numeriek, alfanumeriek, decimaal \\
    3 & Datahierarchie: level \texttt{01}, \texttt{05} en \texttt{77} \\
    4 & Conditionele logica: \texttt{IF}, \texttt{ELSE}, \texttt{END-IF} \\
    5 & Iteraties: \texttt{PERFORM VARYING ... UNTIL} \\
    6 & Paragrafen en \texttt{PERFORM} \\
    7 & Foutafhandeling: \texttt{FILE STATUS} en code \texttt{35} \\
    \bottomrule
  \end{tabularx}
  \vfill
  \footnotesize Levels worden sequentieel ontgrendeld, zodat beginners de programmaskelet beheersen voor ze complexere concepten zien.
\end{frame}

%==============================================================================
\begin{frame}{Gamification in de PoC: Octalysis}
  \small
  \begin{tabularx}{\textwidth}{@{}>{\bfseries}p{5.2cm} X@{}}
    \toprule
    Core Drive & Vertaling in de PoC \\
    \midrule
    Development \& Accomplishment & levels, voortgangsbalk, punten, badges, viering \\
    Empowerment \& Feedback & vrije code-invoer met onmiddellijke validatie \\
    Ownership \& Possession & profiel met avatar, punten en badge-collectie \\
    Social Influence \& Relatedness & vrienden, top drie en volledig leaderboard \\
    Scarcity \& Impatience & sequentieel ontgrendelen, vergrendelde badges \\
    Loss \& Avoidance & bewust beperkt, enkel een first-try bonus \\
    \bottomrule
  \end{tabularx}
  \vfill
  \footnotesize Het ontwerp is bewust \textit{White Hat}-zwaar en mikt op positieve langetermijnmotivatie. Schaarste en verliesangst worden spaarzaam ingezet \autocite{Chou2015}.
\end{frame}

%==============================================================================
\begin{frame}{Waarom dit ontwerp?}
  \footnotesize
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Alternatief A: narratieve gamification}
      \begin{itemize}
        \item spreekt \textit{Epic Meaning} sterk aan
        \item maar rollenspel scoort verwaarloosbaar op motivatie (\(g \approx 0{,}06\)) \autocite{Zhan2022}
        \item duur om te bouwen en te onderhouden
      \end{itemize}
      \medskip
      \textbf{Alternatief B: echte compiler of mainframe}
      \begin{itemize}
        \item realistische foutmeldingen
        \item maar hoge technische complexiteit en tragere feedback
        \item botst met de sandbox-verwachting van junioren
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Gekozen ontwerp}
        Volledig in de browser, geen installatie of mainframe nodig. Regex-checks maken de PoC snel en flexibel. Een Python-voorbeeld staat steeds naast de COBOL-code.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Evaluatie en resultaten}

\begin{frame}{Evaluatie in twee delen}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Deel 1: literatuurevaluatie (uitgevoerd)}
        \footnotesize Elke ontwerpkeuze wordt afgezet tegen empirisch onderzoek. Per keuze: beschrijven, een claim uit de literatuur zoeken en de uitlijning beoordelen.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Deel 2: pilotontwerp (niet uitgevoerd)}
        \footnotesize Een reproduceerbaar experiment: twee groepen leren elk een dag COBOL, een via de PoC en een via een papieren cursus, gevolgd door dezelfde kennistest.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \footnotesize\raggedright
  De literatuurevaluatie levert geen causale claim over \textit{Learn COBOL} zelf, wel een onderbouwde uitspraak over de plausibiliteit van de ontwerpkeuzes.
\end{frame}

%==============================================================================
\begin{frame}{Resultaten van de literatuurevaluatie}
  \begin{itemize}
    \item \textbf{PBL:} punten, badges en leaderboard zijn ingebed in een breder ontwerp en vermijden zo de PBL-fallacy \autocite{KevinWerbach2020}.
    \item \textbf{Octalysis:} de PoC is duidelijk White Hat-zwaar (Core Drives 1 tot 3), met spaarzaam gebruik van schaarste en verliesangst \autocite{Chou2015}.
    \item \textbf{MDA:} checklist, voortgangsbalk en viering laten \textit{Challenge} en beloning op hetzelfde moment samenkomen \autocite{Hunicke2004}.
    \item \textbf{Didactiek:} leerdoel-gerichte feedback en herhaalbare levels sluiten aan bij distributed practice en het spacing effect \autocite{Cepeda2006}.
  \end{itemize}
  \pause
  \begin{block}{Verwacht effect}
    \footnotesize Omdat de PoC de kerncomponenten uit \citeauthor{Zhan2022}~\footfullcite{Zhan2022} implementeert, is een effect in dezelfde grootteorde aannemelijk. Dit blijft een verwachting, er werden geen metingen op deelnemers uitgevoerd.
  \end{block}
\end{frame}

%==============================================================================
\begin{frame}{Pilotontwerp ter validatie}
  \footnotesize
  \begin{itemize}
    \item \textbf{Doelgroep:} junioren met een Python-basis en geen COBOL-ervaring.
    \item \textbf{Steekproef:} \(N = 12\) tot \(20\), ongeveer 6 tot 10 per groep.
    \item \textbf{Opzet:} tussen-proefpersoon, willekeurige toewijzing aan groep A (papieren cursus) of groep B (PoC).
    \item \textbf{Hoofdmeting:} dezelfde kennistest na een leerdag, vergelijking van de groepsgemiddelden.
  \end{itemize}

%==============================================================================
\begin{frame}{Beperkingen}
  \begin{itemize}
    \item \textbf{Geen empirische geldigheid voor de PoC zelf:} de evaluatie ondersteunt plausibiliteit, geen gemeten leerwinst.
    \item \textbf{Selectiebias:} de meta-analyses gaan vooral over moderne talen, niet over COBOL.
    \item \textbf{Confounding:} een deel van het effect komt door interactiviteit op zich, niet enkel door gamification \autocite{Zhan2022}.
    \item \textbf{Heuristische validatie:} regex-checks kunnen false positives en false negatives geven.
  \end{itemize}
\end{frame}

\section{Conclusie}

\begin{frame}{Conclusie en bijdrage}
  \begin{block}{Antwoord op de onderzoeksvraag}
    Op basis van de literatuurevaluatie is het aannemelijk dat de gamification-elementen in \textit{Learn COBOL} een positief effect hebben op motivatie en leerwinst.
  \end{block}
  \textbf{Viervoudige bijdrage}
  \begin{itemize}\small
    \item een theoretisch kader dat gamification koppelt aan mainframe-onderwijs
